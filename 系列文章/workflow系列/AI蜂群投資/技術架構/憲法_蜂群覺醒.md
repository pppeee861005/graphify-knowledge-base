憲法的實體結構
stock-swarm-wf/
├── constitution/
│   ├── schemas.js        # 第一部：數據憲章
│   ├── verification.js   # 第二部:驗證模式
│   └── prompts.js        # 第三部：領域 prompt 庫
└── workflows/
    └── swarm-scan.js     # 肉身（引用憲法）
核心原則：憲法與工作流分離。工作流可以被 ultracode 重寫一百次，憲法不動。

第一部：Schema（數據憲章）
寫法要領：每個 schema 都是一次「強迫 agent 表態」。不允許模糊輸出。
javascript// constitution/schemas.js

// 偵察報告：只准回報事實，不准有意見
export const SCOUT_REPORT = {
  type: 'object',
  properties: {
    ticker:        { type: 'string' },
    triggered:     { type: 'boolean' },       // 強迫二元表態
    volumeRatio:   { type: 'number' },        // 850% → 8.5，要數字不要形容詞
    orderImbalance:{ type: 'number' },        // 5.2:1 → 5.2
    largestOrder:  { type: 'number' },        // 金額，單位萬
    rawFacts:      { type: 'string', maxLength: 200 }  // 限長=禁止小作文
  },
  required: ['ticker', 'triggered', 'volumeRatio']
}

// 研判裁決：強迫從封閉選項中選,不准發明新分類
export const VERDICT = {
  type: 'object',
  properties: {
    classification: { enum: ['機構建倉', '誘多陷阱', '新聞驅動', '噪聲'] },  // 封閉世界
    confidence:     { type: 'number', minimum: 0, maximum: 1 },
    refutation:     { type: 'string' },   // 必填：你嘗試否定自己的理由
    keyEvidence:    { type: 'string' }
  },
  required: ['classification', 'confidence', 'refutation']  // refutation 是必填！
}

// 預警輸出:給人看的，不是給機器看的
export const ALERT = {
  type: 'object',
  properties: {
    ticker:      { type: 'string' },
    action:      { enum: ['觀察', '準備進場', '立即關注'] },
    entryLevel:  { type: 'number' },
    invalidation:{ type: 'number' },   // 什麼價位證明判斷錯了——可證偽性！
    oneLiner:    { type: 'string', maxLength: 100 }
  },
  required: ['ticker', 'action', 'invalidation']
}
三個設計哲學藏在裡面：

enum 是憲法的封閉世界條款——agent 不准發明「可能是機構也可能不是」這種廢話分類
refutation 設為 required——把你的懷疑主義寫進資料結構，agent 不反駁自己就無法通過驗證
invalidation 必填——這是你 F/I/S 框架的可證偽性,直接編譯進 schema


第二部：驗證模式（Harness 憲章）
寫法要領：把驗證寫成可複用的函數，不是散在工作流裡的邏輯。
javascript// constitution/verification.js

// 模式一：對抗性三票制——任何發現需要 2/3 懷疑者無法否定
export async function adversarialVote(finding, { judges = 3, threshold = 2 } = {}) {
  const votes = await parallel(
    Array.from({ length: judges }, (_, i) => () =>
      agent(
        `你是懷疑者 ${i + 1}。你的任務是否定這個判斷，預設它是錯的：${finding.rawFacts}\n` +
        `只有在你窮盡反駁理由仍無法否定時，才回 survives: true。`,
        { phase: 'Verify', schema: {
            type: 'object',
            properties: { survives: { type: 'boolean' }, reason: { type: 'string' } },
            required: ['survives']
        }}
      )
    )
  )
  const surviving = votes.filter(Boolean).filter(v => v.survives).length
  return { passed: surviving >= threshold, votes }
}

// 模式二：視角分歧驗證——三個 judge 用不同鏡片,而非三個相同的
export const LENSES = [
  { key: 'chips',  lens: '從籌碼面反駁：這個大單有沒有可能是出貨而非建倉？' },
  { key: 'macro',  lens: '從總經面反駁：今天有沒有 Fed/CPI 等事件讓全市場都在動？' },
  { key: 'trap',   lens: '從行為金融反駁：這像不像典型的誘多陷阱型態？' },
]

// 模式三：靜默權——低於門檻一律不輸出
export function silenceFilter(verdicts, minConfidence = 0.7) {
  return verdicts.filter(v =>
    v.classification === '機構建倉' && v.confidence >= minConfidence
  )
}
關鍵：模式二比模式一更值錢。三個相同的懷疑者是冗餘，三個不同鏡片的懷疑者是多樣性——籌碼面、總經面、行為金融面，這三個鏡片的選擇就是你的領域知識。ultracode 只會生成三個一模一樣的 verifier。

第三部：領域 Prompt 庫
寫法要領：每個 prompt 都要包含**「業餘者不知道的那句話」**。
javascript// constitution/prompts.js

export const SCOUT_PROMPT = (ticker) => `
檢查 ${ticker} 過去 5 分鐘的盤口數據。
只回報事實，禁止推論。

觸發條件（憲法第一條）：
- 成交量 > 前 30 分鐘均值的 300%
- 委買賣比 > 4:1 或 < 1:4
- 單筆成交 > 1000 萬台幣等值

注意：開盤後 15 分鐘與收盤前 15 分鐘的成交量突增屬於正常現象，
除非伴隨委買賣比失衡，否則不觸發。  ← 業餘者不知道的那句話
`

export const JUDGE_PROMPT = (signal) => `
${signal.ticker} 觸發異動：${signal.rawFacts}

執行研判協議（憲法第二條）：
1. 檢索：今天是財報日嗎？有 SEC 8-K 嗎？有產業新聞嗎？
2. 若無新聞驅動 → 進入陷阱檢測：
   - 大單是「一筆巨量」還是「連續中單」？前者更可能是誘多。 ← 領域知識
   - 異動後 3 分鐘內價格是否回落超過漲幅的 50%？是 → 誘多概率大增。
3. 你必須先寫 refutation（嘗試否定的理由），再給結論。

禁止事項：不准說「建議謹慎觀察」這種沒有資訊量的話。
你必須從封閉選項中選一個，並給出讓判斷可被證偽的價位。
`

實戰打磨：憲法的修正案機制
憲法不是一次寫完的。你需要一個回饋迴路：
javascript// constitution/amendments.md（人工維護的修正記錄）

## 修正案 #1（2026-06-XX）
問題：PLTR 預警後隔天低開，confidence 0.82 的判斷錯了
診斷：judge 沒檢查期權到期日，當天是 OpEx，大單是做市商對沖
修正：JUDGE_PROMPT 加入「檢查是否為期權到期週」
→ 這句話進入憲法，永久生效
每一次實盤錯誤 = 一條修正案 = 憲法密度 +1。 這就是「實戰打磨」的字面意思。六個月後，你的 prompts.js 裡會有二十句業餘者不知道的話——那才是 npm 套件裡真正無法被 fork 複製的東西（他們能複製文字，複製不了背後的虧損學費）。
